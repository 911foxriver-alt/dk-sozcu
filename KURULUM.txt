  // Seçili birim (ya da tüm birimler) için veri içeren yılları sunucudan

  // sorgu atmamak üzere önbelleğe alınır.
  function ensureRoadmapYears(teamId, forceRefresh) {
    const cacheKey = teamId || '__all__';
    if (!forceRefresh && state.roadmapYearsCache[cacheKey]) {
      populateYearSelect(state.roadmapYearsCache[cacheKey]);
      return Promise.resolve(state.roadmapYearsCache[cacheKey]);
    }
    return new Promise((resolve, reject) => {
      google.script.run
        .withSuccessHandler(years => {
          state.roadmapYearsCache[cacheKey] = years || [];
          populateYearSelect(state.roadmapYearsCache[cacheKey]);
          resolve(state.roadmapYearsCache[cacheKey]);
        })
        .withFailureHandler(reject)
        .getAvailableYears(teamId);
    });
  }

  function populateYearSelect(years) {
    const select = document.getElementById('roadmapYearSelect');
    if (!select) return;
    const currentYear = new Date().getFullYear();
    const allYears = Array.from(new Set([...(years || []), currentYear])).sort((a, b) => a - b);
    const prevValue = state.roadmapYear || currentYear;
    select.innerHTML = allYears.map(y => `<option value="${y}">${y}</option>`).join('');
    select.value = allYears.includes(prevValue) ? String(prevValue) : String(currentYear);

  // Aynı sekmeye/tarihe/ekibe kısa süre içinde tekrar dönüldüğünde sunucuya
  // gitmeden anında gösterebilmek için, o an görüntülenen ekranı benzersiz
  // şekilde tanımlayan bir anahtar üretir.
  function cacheKeyForCurrent() {
    return [state.currentView, state.currentTeamId || '', toIsoDate(state.currentDate)].join('|');

  // Önbellekteki bir verinin "taze" sayılacağı süre. Sunucu tarafındaki
  // paylaşımlı önbellek zaten 5 dakikada bir tazeleniyor; burada kısa
  // tutuyoruz ki veri hiçbir zaman gerçekte olduğundan fazla eski görünmesin —
  // sadece art arda sekme değiştirmeler anında olsun.
  const CLIENT_CACHE_TTL_MS = 90 * 1000;

  function loadCurrent(forceRefresh) {
    if (state.currentView === 'roadmap') { loadRoadmap(forceRefresh); return; }

    const key = cacheKeyForCurrent();
    const cached = !forceRefresh && state.cache[key];

    if (forceRefresh || !cached) {
      setLoading();
    } else {
      // Önbellekte veri var: sunucu cevabını beklemeden hemen göster.
      handleLoaded(cached.data);
      if (Date.now() - cached.time < CLIENT_CACHE_TTL_MS) return; // taze: sunucuya hiç gitme
      // taze değil: ekranı bozmadan arka planda sessizce güncelle
    }

    const refreshBtn = document.getElementById('refreshBtn');
    if (refreshBtn) refreshBtn.disabled = true;
    const date = toIsoDate(state.currentDate);
    const finish = data => {
      if (refreshBtn) refreshBtn.disabled = false;
      state.cache[key] = { data: data, time: Date.now() };
      // Cevap gelene kadar kullanıcı başka bir sekmeye geçmiş olabilir;
      // yalnızca hâlâ aynı ekrandaysa görüntüyü bu cevapla güncelle.
      if (cacheKeyForCurrent() === key) handleLoaded(data);
    };
    const fail = error => {
      if (refreshBtn) refreshBtn.disabled = false;
      // Önbellekten zaten bir şey gösterildiyse, arka plan tazelemesi
      // başarısız olsa bile ekranı hataya çevirme.
      if (forceRefresh || !cached) showError(error);
    };
    const runner = google.script.run.withSuccessHandler(finish).withFailureHandler(fail);
    if (state.currentView === 'overview') runner.getOverview(date, !!forceRefresh);
    else if (state.currentView === 'planner') runner.getPlannerWeek(date, !!forceRefresh);
    else if (state.currentView === 'migration') runner.getMigrationStats(!!forceRefresh);
    else if (state.currentView === 'classStudents') runner.getClassStudentStats(!!forceRefresh);
    else if (state.currentView === 'capacity') runner.getCapacityOccupancyStats(!!forceRefresh);
    else if (state.currentView === 'departure') runner.getStudentDepartureStats(!!forceRefresh);
    else if (state.currentView === 'transition') runner.getDaimiyeTransitionStats(!!forceRefresh);
    else if (state.currentView === 'summaryGrowth') runner.getEnsarMuhacirGrowthStats(!!forceRefresh);
    else runner.getTeamActivities(state.currentTeamId, date, !!forceRefresh);
  }

  // Yol Haritası ekranı için ayrı yükleme akışı: seçili moda göre (Yaklaşan
  // Haftalar / Yıla Göre Gözat) doğru sunucu fonksiyonunu çağırır. Yıl modunda
  // önce o bağlam için veri içeren yılların yüklenmiş olduğundan emin olunur.
  // Yol Haritası ekranı için de aynı önbellek mantığı: mod/ekip/tarih/hafta
  // sayısı/yön kombinasyonuna göre benzersiz bir anahtar üretir.
  function cacheKeyForRoadmap() {
    if (state.roadmapMode === 'year') {
      return ['roadmap-year', state.currentTeamId || '', state.roadmapYear || ''].join('|');
    }
    return ['roadmap-weeks', state.currentTeamId || '', toIsoDate(state.currentDate), state.roadmapWeekCount, state.roadmapDirection].join('|');
  }

  function loadRoadmap(forceRefresh) {
    const key = cacheKeyForRoadmap();
    const cached = !forceRefresh && state.cache[key];

    if (forceRefresh || !cached) {
      setLoading();
    } else {
      renderRoadmap(cached.data);
      if (Date.now() - cached.time < CLIENT_CACHE_TTL_MS) return;
    }

    const refreshBtn = document.getElementById('refreshBtn');
    if (refreshBtn) refreshBtn.disabled = true;
    const done = data => {
      if (refreshBtn) refreshBtn.disabled = false;
      state.cache[key] = { data: data, time: Date.now() };
      // Cevap gelene kadar kullanıcı modu/haftayı/yılı değiştirmiş olabilir;
      // yalnızca hâlâ aynı görünüm isteniyorsa ekranı bu cevapla güncelle.
      if (cacheKeyForRoadmap() === key) renderRoadmap(data);
    };
    const fail = error => {
      if (refreshBtn) refreshBtn.disabled = false;
      if (forceRefresh || !cached) showError(error);
    };

    if (state.roadmapMode === 'year') {
      ensureRoadmapYears(state.currentTeamId, forceRefresh).then(() => {
        const year = state.roadmapYear || new Date().getFullYear();
        google.script.run
          .withSuccessHandler(done)
          .withFailureHandler(fail)
          .getRoadmapYear(state.currentTeamId, year, !!forceRefresh);
      }).catch(fail);
      return;
    }

    const date = toIsoDate(state.currentDate);
    google.script.run
      .withSuccessHandler(done)
      .withFailureHandler(fail)
      .getRoadmap(state.currentTeamId, date, state.roadmapWeekCount, !!forceRefresh, state.roadmapDirection);
  }

  function handleLoaded(data) {

    else if (state.currentView === 'departure') renderDeparture(data);
    else if (state.currentView === 'transition') renderTransition(data);
    else if (state.currentView === 'summaryGrowth') renderSummaryGrowth(data);
    else receiveData(data);
  }

  function migrationTotals(items) {
    const totals = items.reduce((acc, item) => {
      acc.muhacir += Number(item.muhacir) || 0;
      acc.ensar += Number(item.ensar) || 0;
      return acc;
    }, { muhacir: 0, ensar: 0 });
    totals.total = totals.muhacir + totals.ensar;
    totals.ratio = totals.total ? totals.muhacir / totals.total * 100 : 0;
    totals.ratioText = formatPercentClient(totals.ratio);
    return totals;
  }

  function formatPercentClient(value) {
    return (Number(value) || 0).toLocaleString('tr-TR', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + '%';
  }

    const t=items.reduce((a,x)=>{a.capacity+=+x.capacity||0;a.november+=+x.november||0;a.current+=+x.current||0;a.needFull+=+x.needFull||0;a.overCapacity+=+x.overCapacity||0;if(x.occupancyBand==='low')a.lowCount++;if(x.occupancyBand==='over')a.overCount++;if(x.growthBand==='down')a.declineCount++;return a;},{capacity:0,november:0,current:0,needFull:0,overCapacity:0,lowCount:0,overCount:0,declineCount:0});

  }

  function renderPlannerEvent(item) {

  function groupByDate(items) { return items.reduce((acc, item) => { const key = item.date || ''; (acc[key] ||= []).push(item); return acc; }, {}); }

  // Bir ekip kaydının hangi gün(ler)de gösterileceğini belirler:
  // - Başlama ve Bitiş tarihi birlikte doluysa, aradaki HER günde gösterilir.
  // - Sadece Başlama tarihi doluysa, o günde gösterilir.
  // - Başlama boş ama Son Teslim Tarihi doluysa, Son Teslim Tarihi gününde gösterilir.
  // - Hiçbiri dolu değilse tarihsiz kabul edilir.
  function datesForTeamItem(item) {
    if (item.date && item.endDate && item.endDate >= item.date) {
      const dates = [];
      let cursor = new Date(item.date + 'T12:00:00');
      const end = new Date(item.endDate + 'T12:00:00');
      while (cursor <= end) {
        dates.push(toIsoDate(cursor));
        cursor.setDate(cursor.getDate() + 1);
      }
      return dates;
    }
    if (item.date) return [item.date];
    if (item.dueDate) return [item.dueDate];
    return [];
  }

  function groupTeamItemsByDay(items) {
    const groups = {};
    const undated = [];
    items.forEach(item => {
      const dates = datesForTeamItem(item);
      if (!dates.length) { undated.push(item); return; }
      dates.forEach(date => { (groups[date] ||= []).push(item); });
    });
    return { groups, undated };
  }

  // Yol Haritası ekranı: tarihsiz/süregelen işler + önümüzdeki N hafta,
  // her biri açılır-kapanır (akordeon) bir grup olarak gösterilir.
  // Tek bir birim seçiliyse kartlar doğrudan listelenir; tüm birimler
  // seçiliyse her hafta grubu içinde ayrıca birim başlıklarına bölünür.
  function renderRoadmap(data) {
    const host = document.getElementById('contentArea');
    const flat = [];
    // Kartlar oluşturulurken renderCard() doğru detay indeksini bulabilsin.
    // Aynı dizi referansını baştan state'e bağlarız; gruplar eklendikçe indeksler güncel kalır.
    state.items = flat;
    const groupsHtml = [];
    const showTeamGroups = !data.team;

    if (data.undated && data.undated.length) {
      const key = 'undated';
      data.undated.forEach(item => flat.push(item));
      const isOpen = state.roadmapOpenState.hasOwnProperty(key) ? state.roadmapOpenState[key] : true;
      groupsHtml.push(renderRoadmapGroup({
        key,
        title: '📌 Tarihsiz / Süregelen İşler',
        subtitle: 'Tarihi belirtilmemiş, her zaman gündemde olan kayıtlar',
        items: data.undated,
        isOpen,
        isCurrent: false,
        showTeamGroups
      }));
    }

    if (data.mode === 'year') {
      const anyCurrentPeriod = (data.periods || []).some(p => state.today >= p.start && state.today <= p.end);
      (data.periods || []).forEach((period, idx) => {
        const key = 'period-' + data.year + '-' + idx;
        period.items.forEach(item => flat.push(item));
        const isCurrentPeriod = state.today >= period.start && state.today <= period.end;
        const defaultOpen = isCurrentPeriod || (!anyCurrentPeriod && idx === 0);
        const isOpen = state.roadmapOpenState.hasOwnProperty(key) ? state.roadmapOpenState[key] : defaultOpen;
        groupsHtml.push(renderRoadmapGroup({
          key,
          title: `${period.label} ${data.year}`,
          subtitle: isCurrentPeriod ? 'İçinde bulunduğunuz dönem' : '',
          items: period.items,
          isOpen,
          isCurrent: isCurrentPeriod,
          showTeamGroups
        }));
      });
    } else {
      (data.weeks || []).forEach(week => {
        const key = 'week-' + week.start;
        week.items.forEach(item => flat.push(item));
        const offset = week.offset || 0;
        const isCurrentWeek = state.today >= week.start && state.today <= week.end;

        let label;
        if (offset === 0) label = 'Bu Hafta';
        else if (offset === 1) label = 'Gelecek Hafta';
        else if (offset === -1) label = 'Geçen Hafta';
        else if (offset > 1) label = `${offset} Hafta Sonra`;
        else label = `${Math.abs(offset)} Hafta Önce`;

        const defaultOpen = offset >= 0 && offset <= 2; // bu hafta + sonraki 2 hafta varsayılan açık
        const isOpen = state.roadmapOpenState.hasOwnProperty(key) ? state.roadmapOpenState[key] : defaultOpen;

        groupsHtml.push(renderRoadmapGroup({
          key,
          title: `${label} · ${formatShortDate(week.start)} – ${formatShortDate(week.end)}`,
          subtitle: isCurrentWeek ? 'İçinde bulunduğunuz hafta' : '',
          items: week.items,
          isOpen,
          isCurrent: isCurrentWeek,
          showTeamGroups
        }));
      });
    }

    host.className = flat.length ? '' : 'empty';
    host.innerHTML = flat.length ? groupsHtml.join('') : 'Seçili aralıkta herhangi bir kayıt bulunmuyor.';
  }

  function renderRoadmapGroup(opts) {
    const { key, title, subtitle, items, isOpen, isCurrent, showTeamGroups } = opts;
    const countClass = !items.length ? 'empty' : (isCurrent ? 'current' : '');
    let bodyHtml;

    if (!items.length) {
      bodyHtml = `<div class="roadmap-empty-week">Bu dönem için kayıt yok.</div>`;
    } else if (showTeamGroups) {
      const byTeam = {};
      const order = [];
      items.forEach(item => {
        const tId = item.teamId || '—';
        if (!byTeam[tId]) { byTeam[tId] = []; order.push(tId); }
        byTeam[tId].push(item);
      });
      bodyHtml = order.map(tId => {
        const group = byTeam[tId];
        const teamName = group[0].teamName || tId;
        const teamIcon = group[0].teamIcon || '🏢';
        return `<div class="roadmap-team-block">
          <div class="roadmap-team-label">${escapeHtml(teamIcon)} ${escapeHtml(teamName)} <small>(${group.length} kayıt)</small></div>
          <div class="cards">${group.map(item => renderCard(item)).join('')}</div>
        </div>`;
      }).join('');
    } else {
      bodyHtml = `<div class="cards">${items.map(item => renderCard(item)).join('')}</div>`;
    }

    return `<div class="roadmap-group${isOpen ? ' open' : ''}" data-key="${escapeAttr(key)}">
      <button class="roadmap-head" onclick="toggleRoadmapGroup('${escapeJs(key)}')">
        <div class="roadmap-head-left">
          <span class="roadmap-chevron">›</span>
          <div><div class="roadmap-title">${escapeHtml(title)}</div>${subtitle ? `<div class="roadmap-sub">${escapeHtml(subtitle)}</div>` : ''}</div>
        </div>
        <span class="roadmap-count ${countClass}">${items.length} kayıt</span>
      </button>
      <div class="roadmap-body">${bodyHtml}</div>
    </div>`;
  }

  function formatShortDate(iso) { return new Intl.DateTimeFormat('tr-TR', { day: '2-digit', month: 'short', year: 'numeric' }).format(new Date(iso + 'T12:00:00')); }
  function formatLongDate(iso) { return new Intl.DateTimeFormat('tr-TR', { weekday: 'long', day: 'numeric', month: 'long' }).format(new Date(iso + 'T12:00:00')); }
  function formatDateRange(start, end) { if (!start) return ''; if (!end || start === end) return formatLongDate(start); return `${formatLongDate(start)} – ${formatLongDate(end)}`; }
  function toIsoDate(date) { const y = date.getFullYear(); const m = String(date.getMonth()+1).padStart(2,'0'); const d = String(date.getDate()).padStart(2,'0'); return `${y}-${m}-${d}`; }
  function statusClass(value) { const t = String(value).toLocaleLowerCase('tr-TR'); if (t.includes('tamam') || t.includes('bitti')) return 'done'; if (t.includes('plan')) return 'planned'; return ''; }
  function priorityClass(value) { return String(value).toLocaleLowerCase('tr-TR').includes('yüksek') ? 'high' : ''; }
  function getDeadlineState(itemOrDueDate) {
    const item = typeof itemOrDueDate === 'object' && itemOrDueDate !== null
      ? itemOrDueDate
      : { dueDate: itemOrDueDate, endDate: '' };

    // Bir işin bitiş tarihi girilmişse tamamlanmış kabul edilir ve son teslim
    // tarihi geçmiş olsa bile gecikmiş olarak işaretlenmez.
    if (item.endDate) return 'completed';
    if (!item.dueDate) return '';

    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const due = new Date(item.dueDate + 'T12:00:00');
    due.setHours(0, 0, 0, 0);
    const diff = Math.round((due - today) / 86400000);

    if (diff < 0) return 'overdue';
    if (diff <= 3) return 'soon';
    return 'safe';
  }
  function getStatusTone(value) {
    const t = String(value || '').toLocaleLowerCase('tr-TR');
    if (t.includes('tamam') || t.includes('bitti') || t.includes('hazır')) return 'done';
    if (t.includes('engel') || t.includes('iptal')) return 'blocked';
    if (t.includes('bekliyor') || t.includes('plan')) return 'waiting';
    if (t.includes('devam') || t.includes('çalış')) return 'progress';
    return '';
  }

  function escapeHtml(value) { return String(value ?? '').replace(/[&<>'"]/g, ch => ({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[ch])); }
  function escapeAttr(value) { return escapeHtml(value); }
